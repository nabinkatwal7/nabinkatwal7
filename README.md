```go
package main

import (
    "fmt"
    "time"
)

type Developer struct {
    Name       string
    Language   string
    OS         string
    Editor     string
    AITools    []string
    Motivation string
}

func (d Developer) Code() {
    for {
        fmt.Printf("[%s] Building something awesome...\n",
            time.Now().Format("15:04:05"))

        time.Sleep(time.Second)
    }
}

func main() {
    me := Developer{
        Name:       "Nabin Katwal",
        Language:   "Go",
        OS:         "Arch Linux",
        Editor:     "Neovim",
        AITools:    []string{"Codex", "Claude", "Cursor"},
        Motivation: "Build. Break. Learn. Repeat.",
    }

    fmt.Printf("Hello, I'm %s!\n", me.Name)

    me.Code()
}
```
